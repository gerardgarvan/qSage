def _compute_N(num_rows, topshift, is_combo):
    """
    Compute number of q-coefficients to use per row (D-04).
    C = 21 for findhom/findnonhom; C = 20 for *combo variants.
    N = num_rows + C + topshift
    """
    C = 20 if is_combo else 21
    return num_rows + C + topshift


def _parse_optional_args(args):
    """
    Dispatch trailing optional arguments (D-05).
    int -> topshift, 'yes'/'no' -> etaoption.
    Returns (topshift, etaoption).
    """
    topshift = 0
    etaoption = 'no'
    for a in args:
        if isinstance(a, int):
            topshift = a
        elif a in ('yes', 'no'):
            etaoption = a
    return topshift, etaoption


def _eval_monomial(exponents, series_list):
    """
    Evaluate a monomial: product of series_list[i].series ** exponents[i].
    Returns a Sage LaurentSeriesRing element.
    All series must share the same parent ring.
    """
    parent = series_list[0].series.parent()
    val = parent.one()
    for s, e in zip(series_list, exponents):
        if e != 0:
            val = val * s.series ** e
    return val


def _validate_series_lengths(L, N, label='L'):
    """Raise ValueError if any series in L has fewer than N terms."""
    for i, s in enumerate(L):
        if s.T < N:
            raise ValueError(
                f"Series {label}[{i}] has precision {s.T} but N={N} "
                f"terms required; recompute with T >= {N}"
            )


def _build_kernel_polys(kern, mono_exps, R):
    """Map a Sage kernel (left_kernel()) to a set of polynomials in R."""
    gens = R.gens()
    result = set()
    for vec in kern.basis():
        poly = R.zero()
        for coeff, exponents in zip(vec, mono_exps):
            if coeff != 0:
                term = R.one()
                for g, e in zip(gens, exponents):
                    term *= g ** e
                poly += coeff * term
        if poly != R.zero():
            result.add(poly)
    return result


def _gcd_list(vals):
    """GCD of a list of integers, initialised to 0 (gcd(0,x)=x)."""
    return reduce(gcd, (abs(int(v)) for v in vals), 0)


def _suppress_implied(triples):
    """
    Remove [B, A, M] if an already-accepted [B2, A2, M2] satisfies:
      A2 | A  and  B % A2 == B2 % A2  and  M2 % M == 0.
    (D-07 implication filter.)
    """
    accepted = []
    for B, A, M in triples:
        implied = False
        for B2, A2, M2 in accepted:
            if A % A2 == 0 and B % A2 == B2 % A2 and M2 % M == 0:
                implied = True
                break
        if not implied:
            accepted.append((B, A, M))
    return accepted
