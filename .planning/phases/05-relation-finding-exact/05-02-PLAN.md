```python
    topshift, etaoption = _parse_optional_args(args)
    k = len(L)
    # Homogeneous degree exactly n in k variables
    mono_exps = list(IntegerVectors(n, k))  # each entry sums to n
```

```python
def findhomcombo(f, L, q, n, *args):
    topshift, etaoption = _parse_optional_args(args)
    k = len(L)
    mono_exps = list(IntegerVectors(n, k))
    num_rows = len(mono_exps) + 1  # +1 for the f row
    N = _compute_N(num_rows, topshift, is_combo=True)  # C=20
    _validate_series_lengths(L, N)
    if f.T < N:
        raise ValueError(f"Series f has precision {f.T} but N={N} terms required")
    print(f"# of terms , {N}")
    print(f"matrix is , {num_rows}, x, {N}")

    rows = []
    for exponents in mono_exps:
        val = _eval_monomial(exponents, L)
        rows.append([QQ(val[j]) for j in range(N)])
    # f row last
    rows.append([QQ(f.series[j]) for j in range(N)])

    M = matrix(QQ, rows)
    kern = M.left_kernel()
    R = PolynomialRing(QQ, ['X%d' % (i + 1) for i in range(k)])

    # The f row's coefficient must be non-zero; normalise so f-coefficient = 1
    result = set()
    for vec in kern.basis():
        f_coeff = vec[-1]
        if f_coeff == 0:
            continue
        poly = R.zero()
        for coeff, exponents in zip(vec[:-1], mono_exps):
            if coeff != 0:
                term = R.one()
                gens = R.gens()
                for g, e in zip(gens, exponents):
                    term *= g ** e
                poly += (coeff / (-f_coeff)) * term
        if poly != R.zero():
            result.add(poly)

    if etaoption == 'yes':
        _print_etaoption(result, L, q)
    return result
```

```python
def _print_etaoption(result, L, q):
    """When etaoption='yes', print each polynomial with each monomial's etamake form."""
    T = min(s.T for s in L)
    for poly in result:
        print(poly)
        # Best-effort: evaluate each monomial term at L, call etamake
        # (print only; not part of return value)
        try:
            from qseries._conversion import etamake
            R = poly.parent()
            gens = R.gens()
            for mono, coeff in poly.dict().items():
                val = L[0].series.parent().one()
                for s, e in zip(L, mono):
                    val = val * s.series ** e
                from qseries._series import QSeries
                try:
                    ep = etamake(QSeries(val, T), q, T)
                    print(f"  {coeff} * {ep}")
                except Exception:
                    pass
        except Exception:
            pass
```

findlincombo is a special case of findhomcombo with n=1 (degree exactly 1). It builds the k monomials [X1, X2, ..., Xk] (each has exponent list with a 1 in position i and 0 elsewhere), adds an f row, and uses the same left_kernel mechanism.

The C constant for findlincombo: from D-04, "degree 1 for findlincombo" and "combo variants C=20". So num_rows = k+1 (k basis rows + f row), N = (k+1) + 20 + topshift.

Wait — re-read D-04 more carefully: N = (#rows) + C + topshift. For findlincombo, the "rows" count is k monomials (degree-1 basis, same as IntegerVectors(1, k) which gives k unit vectors) + 1 f row = k+1. C=20 (combo). N = (k+1) + 20 + topshift. This is consistent with "degree 1 for findlincombo" and C=20.

```python
def findlincombo(f, L, SL, q, topshift=0):
    k = len(L)
    # Linear monomials: IntegerVectors(1, k) gives k unit-vector tuples
    mono_exps = list(IntegerVectors(1, k))  # [[1,0,...,0], [0,1,...,0], ...]
    num_rows = len(mono_exps) + 1  # k basis rows + f row
    N = _compute_N(num_rows, topshift, is_combo=True)  # C=20
    _validate_series_lengths(L, N)
    if f.T < N:
        raise ValueError(f"Series f has precision {f.T} but N={N} terms required")
    print(f"# of terms , {N}")
    print(f"matrix is , {num_rows}, x, {N}")

    rows = []
    for exponents in mono_exps:
        val = _eval_monomial(exponents, L)
        rows.append([QQ(val[j]) for j in range(N)])
    rows.append([QQ(f.series[j]) for j in range(N)])

    M = matrix(QQ, rows)
    kern = M.left_kernel()
    R = PolynomialRing(QQ, ['X%d' % (i + 1) for i in range(k)])

    result = set()
    for vec in kern.basis():
        f_coeff = vec[-1]
        if f_coeff == 0:
            continue
        poly = R.zero()
        gens = R.gens()
        for coeff, exponents in zip(vec[:-1], mono_exps):
            if coeff != 0:
                # degree-1 monomial: just g_i
                for g, e in zip(gens, exponents):
                    if e == 1:
                        poly += (coeff / (-f_coeff)) * g
        if poly != R.zero():
            result.add(poly)

    if not result:
        print("%NOT A LINEAR COMBO.%")
    return result
```
