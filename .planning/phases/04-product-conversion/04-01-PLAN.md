    ```python
    from sage.all import QQ
    from qseries._prodmake import prodmake
    from qseries._building_blocks import etaq
    from qseries._symbolic import EtaProduct, EProduct, QProduct
    from qseries._series import QSeries

    def etamake(f, q, T):
        """Greedy eta-product identification (CONV-02, D-04/D-05).
        Oracle: etamake(theta3(q,100),q,100) -> eta(2tau)^5/(eta(4tau)^2*eta(tau)^2)
        Oracle: etamake(theta4(q,100),q,100) -> eta(tau)^2/eta(2tau)
        Oracle: etamake(theta2(q,100),q,100) -> 2*eta(4tau)^2/eta(2tau) [r per D-04 step 5]
        """
        # Step 1: strip fractional q-shift and leading constant
        r_input = QQ(f.r)
        # Work on f.series (Laurent series). leading coeff = f.series[0]
        c = QQ(f.series[0])
        # Normalize: divide by c. We need a QSeries with constant term 1.
        R = f.series.parent()
        residual_series = (f.series / c).add_bigoh(T)
        # Wrap as QSeries with r=0
        from qseries._series import QSeries as QS
        residual = QS(residual_series, T=T)  # r=0, b0=1

        factors = []  # list of (level, exponent) pairs

        # Greedy loop: eliminate lowest non-unit term
        for _ in range(T - 1):
            # Find the lowest k >= 1 where coefficient != 0
            k = None
            for n in range(1, T):
                coeff = QQ(residual.series[n])
                if coeff != 0:
                    k = n
                    c_k = coeff
                    break
            if k is None:
                break  # residual == 1 + O(q^T)

            # Multiply residual by etaq(q,k,T)^(c_k) to cancel +c_k*q^k term.
            # This works because etaq(q,k,T) = 1 - q^k + ..., so (1-q^k+...)^(c_k)
            # contributes -c_k*q^k at leading order, cancelling +c_k*q^k.
            # Record eta(k*tau)^(-c_k) in factors (opposite sign, per D-04 step 3).
            eta_k = etaq(q, k, T)
            c_k_q = QQ(c_k)
            if c_k_q.denominator() != 1:
                raise ValueError(f"Non-integer leading coefficient {c_k} at q^{k}; cannot cancel exactly")
            c_k_int = int(c_k_q)
            # residual *= eta_k^(c_k_int)
            if c_k_int >= 0:
                multiplier = eta_k ** c_k_int
            else:
                multiplier = eta_k ** (-c_k_int)
                # need reciprocal — use series inversion
                inv_series = multiplier.series.__invert__().add_bigoh(T)
                multiplier = QS(inv_series, T=T)
            residual = residual * multiplier
            # Record factor: eta(k*tau)^(-c_k_int)
            factors.append((k, -c_k_int))

        # Step 5: compute EtaProduct.r
        # Each eta(k*tau) carries a q^(k/24) factor in the full modular form sense.
        # Total eta shift = sum(k * e_k / 24) for (k, e_k) in factors.
        eta_shift = sum(QQ(k) * QQ(e) / QQ(24) for k, e in factors)
        r_out = r_input - eta_shift

        # Merge duplicate levels in factors
        from collections import defaultdict
        merged = defaultdict(QQ)
        for k, e in factors:
            merged[k] += QQ(e)
        factors_merged = [(k, int(e)) for k, e in sorted(merged.items()) if e != 0]

        return EtaProduct(constant=c, r=r_out, factors=factors_merged)
    ```

    ```python
        """Greedy _E-product identification (CONV-03).
        Oracle: qetamake(triangular_series, q, 30) -> EProduct({1:-1, 2:2})
        """
        from sage.all import QQ
        from qseries._building_blocks import etaq
        from qseries._symbolic import EProduct
        from qseries._series import QSeries as QS
        from collections import defaultdict

        if QQ(f.r) != 0:
            raise ValueError(f"qetamake requires r=0 (no fractional q-shift), got r={f.r}")
        c = QQ(f.series[0])
        residual_series = (f.series / c).add_bigoh(T)
        residual = QS(residual_series, T=T)

        factors = []
        for _ in range(T - 1):
            k = None
            for n in range(1, T):
                coeff = QQ(residual.series[n])
                if coeff != 0:
                    k = n
                    c_k = coeff
                    break
            if k is None:
                break
            c_k_q = QQ(c_k)
            if c_k_q.denominator() != 1:
                raise ValueError(f"qetamake: non-integer coefficient {c_k} at q^{k}")
            c_k_int = int(c_k_q)
            eta_k = etaq(q, k, T)
            if c_k_int >= 0:
                multiplier = eta_k ** c_k_int
            else:
                multiplier = QS(eta_k.series.__invert__().add_bigoh(T) ** (-c_k_int), T=T)
            residual = residual * multiplier
            factors.append((k, -c_k_int))

        merged = defaultdict(QQ)
        for k, e in factors:
            merged[k] += QQ(e)
        exponents = {k: int(e) for k, e in sorted(merged.items()) if e != 0}
        return EProduct(exponents=exponents)
    ```
