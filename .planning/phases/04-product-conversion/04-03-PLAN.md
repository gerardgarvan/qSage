def checkprod(f, M, Q):
    """Check if a q-series has a 'nice' infinite product representation.
    f:  QSeries with constant term 1 and r=0
    M:  maximum allowed absolute value of any exponent e_n  (integer, NOT None)
    Q:  number of exponents to compute (= T argument to prodmake)
    Returns [c, exponent_list] if all |e_n| <= M, else False.
    """

    from dataclasses import dataclass, field
    from typing import List, Dict, Any
    from collections import namedtuple

    @dataclass
    class MProduct:
        """Return type of mprodmake: product of (1+q^n) factors.
        exponents: list of n values where (1+q^n) appears (multiplicity tracked).
        """
        exponents: List[int] = field(default_factory=list)

        def __repr__(self):
            parts = []
            for n in sorted(self.exponents):
                parts.append("(1+q)" if n == 1 else f"(1+q^{n})")
            return " * ".join(parts) if parts else "1"

        def to_series(self, T):
            from sage.all import LaurentSeriesRing, QQ
            from qseries._series import QSeries
            R = LaurentSeriesRing(QQ, 'q')
            q = R.gen()
            result = R(1).add_bigoh(T)
            for n in self.exponents:
                result = (result * (1 + q**n)).add_bigoh(T)
            return QSeries(result, T=T)

    ```python
    def mprodmake(f, q, T):
        """Greedy (1+q^n) product identification (CONV-05).
        Oracle: mprodmake(etaq(q,2,18)^2/etaq(q,4,18)/etaq(q,1,18), q, 17)
             -> (1+q)(1+q^3)(1+q^5)...(1+q^17)
        Algorithm: same greedy structure as etamake but multiply residual by (1+q^k)^(-c_k).
        """
        from sage.all import QQ, LaurentSeriesRing
        from qseries._series import QSeries as QS

        if f.r != 0:
            raise ValueError(f"mprodmake: input has fractional q-shift r={f.r}")
        c = QQ(f.series[0])
        R = f.series.parent()
        residual_series = (f.series / c).add_bigoh(T)
        residual = QS(residual_series, T=T)

        exponents = []
        for _ in range(T - 1):
            k = None
            for n in range(1, T):
                coeff = QQ(residual.series[n])
                if coeff != 0:
                    k = n
                    c_k = coeff
                    break
            if k is None:
                break
            c_k_q = QQ(c_k)
            if c_k_q.denominator() != 1:
                raise ValueError(f"mprodmake: non-integer coefficient {c_k} at q^{k}")
            c_k_int = int(c_k_q)
            # Multiply by (1+q^k)^(c_k) to cancel: (1+q^k)^(c_k) has leading term +c_k*q^k
            # which cancels the +c_k*q^k in residual. Record (1+q^k)^(-c_k) -> append k (-c_k) times.
            # Build (1+q^k) series and raise to c_k_int
            R2 = residual.series.parent()
            q_gen = R2.gen()
            factor_series = (1 + q_gen**k).add_bigoh(T)
            if c_k_int >= 0:
                mult_series = (factor_series ** c_k_int).add_bigoh(T)
            else:
                mult_series = (factor_series ** (-c_k_int)).add_bigoh(T).__invert__().add_bigoh(T)
            new_res = (residual.series * mult_series).add_bigoh(T)
            residual = QS(new_res, T=T)
            # Record: the output factor (1+q^k)^(-c_k_int)
            # Positive c_k_int: we divided by (1+q^k)^c_k -> product gains (1+q^k)^c_k
            for _ in range(abs(c_k_int)):
                exponents.append(k)

        return MProduct(exponents=exponents)
    ```

    3. Implement `qfactor(f, T=None)`:
    ```python
    def qfactor(f, T=None):
        """Factor polynomial/rational function in QQ[q] into (1-q^k) terms (CONV-06, D-06/D-07).
        Oracle: qfactor(q^6*(1-q^9)*(1-q^10)*(1-q^11)*(1-q^16)/((1-q)*(1-q^2)*(1-q^3)*(1-q^4)))
             -> QFactorResult(numerator=QProduct({9:1,10:1,11:1,16:1}),
                              denominator=QProduct({1:1,2:1,3:1,4:1}),
                              r=6)
        f must be in PolynomialRing(QQ,'q') or its fraction field.
        """
        from sage.all import QQ, ZZ, PolynomialRing
        from qseries._symbolic import QProduct

        # Handle fraction field element: separate numerator and denominator
        try:
            num_poly = f.numerator()
            den_poly = f.denominator()
        except AttributeError:
            # f is a polynomial, not a fraction
            num_poly = f
            den_poly = f.parent()(1)

        def factor_poly(poly):
            """Factor a polynomial, return (r, {k: exp}) for (1-q^k) factors and q^r monomial."""
            r = QQ(0)
            exponents = {}
            factorization = poly.factor()
            unit = QQ(factorization.unit())  # scalar

            for factor, exp in factorization:
                # Check if this is q^1 (a pure monomial)
                if factor == factor.parent().gen():
                    r += QQ(exp)
                elif factor.degree() >= 1:
                    # Check if factor is of form (1 - q^k): degree k, constant=1, leading=-1
                    deg = factor.degree()
                    coeffs = factor.list()  # coeffs[i] = coeff of q^i
                    if (len(coeffs) == deg + 1 and
                            QQ(coeffs[0]) == 1 and QQ(coeffs[deg]) == -1 and
                            all(QQ(coeffs[i]) == 0 for i in range(1, deg))):
                        exponents[deg] = exponents.get(deg, 0) + exp
                    else:
                        # Not a standard (1-q^k) factor — include as-is (best effort)
                        # For now, skip non-cyclotomic factors silently
                        pass
            return r, exponents, unit

        num_r, num_exp, num_unit = factor_poly(num_poly)
        den_r, den_exp, den_unit = factor_poly(den_poly)
        total_r = num_r - den_r

        return QFactorResult(
            numerator=QProduct(exponents=num_exp),
            denominator=QProduct(exponents=den_exp),
            r=total_r
        )
    ```

    def qs2jaccombo(f, q, T, P=None):
        """Convert each term in f to a JacProduct via jacprodmake (CONV-10).
        f: a list of QSeries (one per additive term), or a single QSeries.
        Returns: list of JacProduct objects (one per term).
        If any term cannot be converted, raises ValueError.
        """
        if isinstance(f, list):
            terms = f
        else:
            terms = [f]  # single term

        result = []
        for term in terms:
            if P is not None:
                jp = jacprodmake(term, q, T, base=P)
            else:
                jp = jacprodmake(term, q, T)
            result.append(jp)
        return result
    ```

    def findprod(FL, T, M, Q):
        """Exhaustive search for linear combinations of FL that form nice q-products (CONV-11).
        FL: list of QSeries
        T: max |coefficient| in combinations (search range -T..T per series)
        M: max |exponent| in product — passed directly to checkprod(combo, M, Q)
        Q: highest power of q to consider (truncation for comparison)
        Returns: list of coefficient_lists for each found combination.
        Algorithm: exhaustive search over coefficient tuples. (04-RESEARCH.md D-09)
        """
        from qseries._utilities import checkprod
        from itertools import product as itertools_product
        from sage.all import QQ, LaurentSeriesRing
        from qseries._series import QSeries as QS

        found = []
        n = len(FL)
        R = FL[0].series.parent()
        coeff_range = range(-T, T + 1)
        for coeffs in itertools_product(coeff_range, repeat=n):
            if all(c == 0 for c in coeffs):
                continue
            # Build linear combination
            combo_series = R(0).add_bigoh(Q)
            combo_r = FL[0].r
            for i, c in enumerate(coeffs):
                if c == 0:
                    continue
                term = FL[i]
                scaled = (R(QQ(c)) * term.series).add_bigoh(Q)
                combo_series = (combo_series + scaled).add_bigoh(Q)
            # combo must have r=0 and leading coeff 1 for checkprod
            if combo_series.is_zero():
                continue
            leading = QQ(combo_series[combo_series.valuation()])
            if leading == 0:
                continue
            normed = (combo_series / leading).add_bigoh(Q)
            combo = QS(normed, T=Q)
            # checkprod(f, M, Q) — pass M from findprod's M parameter
            try:
                check_result = checkprod(combo, M, Q)
                if check_result is not False:
                    found.append(list(coeffs))
            except Exception:
                pass  # checkprod may raise on edge cases; skip

        return found
    ```
