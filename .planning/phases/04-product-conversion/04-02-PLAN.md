```python
def _find_period(exponents, T):
    """Smallest P such that e_n == e_{n % P} for all n in exponents with n < T."""
    max_n = max(exponents.keys()) if exponents else 1
    for P in range(1, max_n + 1):
        if all(exponents.get(n, 0) == exponents.get(n % P, 0)
               for n in range(1, T)):
            return P
    return max_n
```

    ```python
    def jacprodmake(f, q, T, base=None):
        """Period detection on prodmake exponents -> JacProduct (CONV-04, D-08).
        Oracle: jacprodmake(tripleprod(q,q^5,10), q, 10) -> JAC(1,5,inf)
        Mandatory: jacprodmake(sqrt(JAC(7,14)/JAC(0,14)), q, 40) -> factor (7,14,1/2)
        base: if given, search only divisors of base for the period P.
        """
        from sage.all import QQ, divisors as sage_divisors

        # Step 1: strip prefix and constant (same as etamake preamble)
        r_input = QQ(f.r)
        c = QQ(f.series[0])
        R = f.series.parent()
        normalized = (f.series / c).add_bigoh(T)
        from qseries._series import QSeries as QS
        f_norm = QS(normalized, T=T)
        # prodmake requires r==0 already; if r_input != 0 we cannot pass to prodmake
        # (fractional prefix means series has no integer-index Laurent expansion starting at 0)
        if r_input != 0:
            raise ValueError(f"jacprodmake: input has fractional q-shift r={r_input}; strip it first")

        # Step 2: call prodmake to get exponent dict {n: e_n}
        qprod = prodmake(f_norm, q, T)
        exponents = qprod.exponents  # {n: e_n} nonzero only

        # Step 3: find smallest period P
        if base is not None:
            # Search divisors of base in increasing order (D-08 / pitfall 3)
            candidates = sorted(sage_divisors(int(base)))
        else:
            max_n = max(exponents.keys()) if exponents else 1
            candidates = range(1, max_n + 1)

        P = None
        for candidate in candidates:
            if all(exponents.get(n, 0) == exponents.get(n % candidate, 0)
                   for n in range(1, T)):
                P = candidate
                break
        if P is None:
            P = max(exponents.keys()) if exponents else 1

        # Step 4: build JacProduct factors from residue classes
        # e_0 is exponents.get(0, 0) but prodmake starts at n=1; skip n=0.
        # Residue classes for a in 0..P-1:
        #   a==0: JAC(0,P,inf)^(e_P) — but n=0 is not tracked; skip or handle
        #   0 < a < P/2: paired with P-a; if e_a == e_{P-a} -> JAC(a,P,inf)^(e_a)
        #               if e_a != e_{P-a} -> JAC(a,P,inf)^(e_a/???) (rational case)
        #   a == P//2 (P even): e_{P/2} -> JAC(P/2,P,inf)^(e_{P//2}/2)

        factors = []
        # Handle a=0: e_0 from n=P (the "period" itself) — actually a=0 means n=kP
        e_0 = exponents.get(P, 0)  # exponent at n=P is the JAC(0,P) exponent
        if e_0 != 0:
            factors.append((0, P, QQ(e_0)))

        seen = set()
        for a in range(1, P):
            if a in seen:
                continue
            paired = P - a
            seen.add(a)
            seen.add(paired)
            e_a = QQ(exponents.get(a, 0))
            e_pa = QQ(exponents.get(paired, 0))
            if a == paired:  # a == P/2
                # JAC(a, P, inf)^(e_a / 2) — half exponent (D-08 mandatory case)
                if e_a != 0:
                    factors.append((a, P, e_a / 2))
            else:
                if e_a == e_pa:
                    if e_a != 0:
                        factors.append((a, P, e_a))
                else:
                    # Unequal paired residues — rational exponents
                    # JAC(a,P,inf)^(e_a/2) * JAC(P-a,P,inf)^(e_pa/2) approximation
                    # This is a simplification; exact treatment requires sqrt-factoring
                    # For now: record both factors with exponent halved (general rational case)
                    if e_a != 0:
                        factors.append((a, P, QQ(e_a)))
                    if e_pa != 0:
                        factors.append((paired, P, QQ(e_pa)))

        return JacProduct(constant=c, r=r_input, factors=factors)
    ```

    ```python
    def jac2prod(jacexpr):
        """JacProduct -> PochProduct via Garvan jac2prod doc page rules (CONV-08).
        JAC(a,b,inf) with a!=0 -> (q^a;q^b)_inf * (q^{b-a};q^b)_inf * (q^b;q^b)_inf
        JAC(0,b,inf)            -> (q^b;q^b)_inf only
        Oracle: jac2prod(JAC(1,5,inf)) -> (q;q^5)*(q^4;q^5)*(q^5;q^5)
        Oracle: jac2prod(JAC(0,10,inf)) -> (q^10;q^10)
        """
        from qseries._symbolic import PochProduct
        factors = []
        for a, b, e in jacexpr.factors:
            e_int = int(QQ(e))  # e must be integer for jac2prod
            if a == 0:
                factors.append((b, b, e_int))
            else:
                factors.append((a, b, e_int))
                factors.append((b - a, b, e_int))
                factors.append((b, b, e_int))
        return PochProduct(factors=factors)
    ```

    ```python
    def jac2series(jacexpr, T):
        """JacProduct -> QSeries via theta substitution (CONV-09).
        JAC(a, b, inf) -> theta(-q^{(b-2a)/2}, q^{b/2}, T)
        Oracle: jac2series(JAC(-4,5,inf), 20) -> (1-q-q^4-q^7-q^13-q^18)*q^(-4) + O(q^16)
        Reference: Garvan jac2series doc page + jac2series substitution formula.
        """
        from sage.all import QQ, LaurentSeriesRing
        from qseries._series import QSeries

        # Start with constant * q^r
        R = LaurentSeriesRing(QQ, 'q')
        q_sym = R.gen()
        result_series = R(QQ(jacexpr.constant)).add_bigoh(T)
        total_r = QQ(jacexpr.r)

        for a, b, e in jacexpr.factors:
            a_q = QQ(a)
            b_q = QQ(b)
            e_q = QQ(e)
            # z = -q^{(b-2a)/2}; this is a Laurent monomial in q with shift (b-2a)/2
            # theta(z, q^{b/2}, T) where q^{b/2} means we substitute q -> q^{b/2}
            # We need to evaluate theta at the level of series.
            # Build z as a QSeries: z = -q^{(b-2a)/2}
            # and the "step" q-parameter as q^{b/2}.
            # In practice: compute theta_series = sum_{i=-T}^{T} z^i * (q^{b/2})^{i^2}
            #   = sum_{i} (-1)^i * q^{i*(b-2a)/2} * q^{i^2*b/2}
            #   = sum_{i} (-1)^i * q^{i*(b-2a)/2 + i^2*b/2}
            # The exponent of q for term i: i*(b-2a)/2 + i^2*b/2
            # This is a Laurent polynomial in q (rational exponents possible).
            # Build directly as a Laurent series with rational shifts.
            shift = (b_q - 2 * a_q) / 2   # z-exponent base = (b-2a)/2
            step = b_q / 2                 # q^(b/2) step

            # Build theta as Laurent series by summing terms
            # sum for i from -N to N where N = isqrt(T) + 1
            from math import isqrt
            N = isqrt(T) + 1
            # Collect terms: exponent of q = i*shift + i^2*step, coefficient = (-1)^i
            # Since exponents may be rational, we accumulate in a dict then build series.
            # To return a QSeries we need to track the minimum exponent as 'r' and
            # build an integer-indexed series for the fractional part.
            # This is complex; instead use the _building_blocks.theta function IF z and q
            # are in the right ring. But theta(z, q, T) expects z as a bivariate element.
            #
            # Simplest correct approach: compute directly.
            # Find min exponent to set as r, then build series with shifted indices.
            terms = {}
            for i in range(-N, N + 1):
                exp = QQ(i) * shift + QQ(i * i) * step
                coeff = (-1) ** i
                terms[exp] = terms.get(exp, QQ(0)) + QQ(coeff)

            min_exp = min(terms.keys())
            # Build as integer-indexed series with shift min_exp
            R2 = LaurentSeriesRing(QQ, 'q')
            q2 = R2.gen()
            s = R2(0).add_bigoh(T)
            for exp, coeff in terms.items():
                int_idx = exp - min_exp
                if int_idx.denominator() != 1:
                    raise ValueError(f"Non-integer index {int_idx} in jac2series; rational shift not integer-reducible")
                if int(int_idx) < T:
                    s = s + R2(coeff) * q2 ** int(int_idx)
            s = s.add_bigoh(T)
            theta_qs = QSeries(s, T=T, r=min_exp)

            # Multiply into result (handling r-shifts and exponents)
            # result = result * theta_qs^e
            if e_q.denominator() != 1:
                raise ValueError(f"jac2series: non-integer exponent {e_q} not supported")
            e_int = int(e_q)
            if e_int >= 0:
                for _ in range(e_int):
                    result_series = (result_series * theta_qs.series).add_bigoh(T)
                    total_r += theta_qs.r
            else:
                for _ in range(-e_int):
                    result_series = (result_series / theta_qs.series).add_bigoh(T)
                    total_r -= theta_qs.r

        return QSeries(result_series, T=T, r=total_r)
    ```

    Note: The direct computation approach avoids the bivariate theta complication.
    The jac2series doc page formula `theta(-q^{(b-2a)/2}, q^{b/2}, T)` is the sum
    `sum_{i=-N}^N (-1)^i * (q^{(b-2a)/2})^i * (q^{b/2})^{i^2}` which equals
